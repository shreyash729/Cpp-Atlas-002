cmake_minimum_required(VERSION 3.16)
project(trie_studio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(trie_service src/service.cpp)
target_include_directories(trie_service PUBLIC include)
add_executable(server src/main.cpp)
target_link_libraries(server PRIVATE trie_service)
add_executable(challenge_tests tests/test_service.cpp)
target_link_libraries(challenge_tests PRIVATE trie_service)
